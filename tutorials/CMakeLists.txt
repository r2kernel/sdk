cmake_minimum_required(VERSION 3.20)
project(r2kernel_tutorials LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)

set(R2KERNEL_CPP_SOURCE_DIR "" CACHE PATH "Optional SDK source directory; otherwise use the configured r2kernel_cpp package.")
if(NOT TARGET r2kernel::cpp)
    if(R2KERNEL_CPP_SOURCE_DIR)
        add_subdirectory("${R2KERNEL_CPP_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
    else()
        find_package(r2kernel_cpp CONFIG REQUIRED)
    endif()
endif()

add_library(r2kernel_tutorial_support INTERFACE)
add_library(r2kernel::tutorial_support ALIAS r2kernel_tutorial_support)
target_compile_features(r2kernel_tutorial_support INTERFACE cxx_std_17)
target_include_directories(r2kernel_tutorial_support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/support/cpp")
target_link_libraries(r2kernel_tutorial_support INTERFACE r2kernel::cpp Threads::Threads)
set_target_properties(r2kernel_tutorial_support PROPERTIES R2KERNEL_TUTORIAL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

set(R2KERNEL_TUTORIAL_FIXTURE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" CACHE PATH
    "Declared fixture root containing transport/; public exports pass <export>/tools/qa/fixtures explicitly.")
set(R2KERNEL_TUTORIAL_FIXTURE_GROUP "all" CACHE STRING "Transport fixture group: all, event, procedure, operation, control, control_drive.")
set_property(CACHE R2KERNEL_TUTORIAL_FIXTURE_GROUP PROPERTY STRINGS all event procedure operation control control_drive)
if(NOT R2KERNEL_TUTORIAL_FIXTURE_GROUP MATCHES "^(all|event|procedure|operation|control|control_drive)$")
    message(FATAL_ERROR "Unknown Tutorial fixture group: ${R2KERNEL_TUTORIAL_FIXTURE_GROUP}")
endif()
set(R2KERNEL_TUTORIAL_OUTPUT_DIR "" CACHE PATH "Optional flat executable output directory for a single fixture group.")
if(R2KERNEL_TUTORIAL_OUTPUT_DIR AND R2KERNEL_TUTORIAL_FIXTURE_GROUP STREQUAL "all")
    message(FATAL_ERROR "A flat Tutorial output directory requires one fixture group (groups share binary names).")
endif()
add_custom_target(r2kernel_tutorial_fixtures)
function(r2kernel_add_transport_fixture group)
    if(NOT R2KERNEL_TUTORIAL_FIXTURE_GROUP STREQUAL "all" AND NOT R2KERNEL_TUTORIAL_FIXTURE_GROUP STREQUAL "${group}")
        return()
    endif()
    foreach(binary IN LISTS ARGN)
        set(target "r2kernel_tutorial_${group}_${binary}")
        add_executable(${target} "${R2KERNEL_TUTORIAL_FIXTURE_ROOT}/transport/${group}/cpp/${binary}.cpp")
        target_link_libraries(${target} PRIVATE r2kernel::tutorial_support)
        set(output "${CMAKE_CURRENT_BINARY_DIR}/bin/${group}")
        if(R2KERNEL_TUTORIAL_OUTPUT_DIR)
            set(output "${R2KERNEL_TUTORIAL_OUTPUT_DIR}")
        endif()
        set_target_properties(${target} PROPERTIES OUTPUT_NAME "${binary}" RUNTIME_OUTPUT_DIRECTORY "${output}")
        foreach(config IN ITEMS DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
            set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${output}")
        endforeach()
        if(APPLE)
            set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "@loader_path")
        elseif(UNIX)
            set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "$ORIGIN")
        endif()
        add_dependencies(r2kernel_tutorial_fixtures ${target})
    endforeach()
endfunction()

r2kernel_add_transport_fixture(event publisher subscriber fanout_subscriber ping echo)
r2kernel_add_transport_fixture(procedure server client)
r2kernel_add_transport_fixture(operation server client)
r2kernel_add_transport_fixture(control gateway controller)
r2kernel_add_transport_fixture(control_drive gateway controller)

    target_compile_features(r2kernel_tutorial_start_gate_test PRIVATE cxx_std_17)

endif()
